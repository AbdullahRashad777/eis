How EIS Auto-Substitution & Rescheduling Works:
1. Same-Subject Priority: Assigns a free teacher from the same division & subject first.
2. Smart Period Swap: If no same-subject teacher is free at Period P, swaps Period P with another period of that class whose teacher is free now.
3. Standby Coverage: Otherwise assigns the division teacher with the lowest daily load & dispatches instant notifications.

-- Emerald International School (EIS) — Smart Behavioral & Academic Database Schema
CREATE TABLE teachers (
  id VARCHAR(32) PRIMARY KEY,
  name VARCHAR(120) NOT NULL,
  email VARCHAR(150) UNIQUE NOT NULL,
  phone VARCHAR(30) NOT NULL,
  division VARCHAR(20) CHECK (division IN ('National', 'American')),
  stage VARCHAR(80) NOT NULL,
  subject VARCHAR(60) NOT NULL,
  max_periods_per_day INT DEFAULT 6,
  emerald_gems INT DEFAULT 120,
  status VARCHAR(20) DEFAULT 'Present'
);

CREATE TABLE students (
  id VARCHAR(32) PRIMARY KEY,
  name VARCHAR(120) NOT NULL,
  division VARCHAR(20) CHECK (division IN ('National', 'American')),
  class_id VARCHAR(32) NOT NULL,
  learning_style VARCHAR(80) NOT NULL,
  emerald_gems INT DEFAULT 100,
  mastery_pct INT DEFAULT 84,
  parent_name VARCHAR(120) NOT NULL,
  parent_phone VARCHAR(30) NOT NULL,
  parent_email VARCHAR(150) NOT NULL,
  parent_link_code VARCHAR(32) UNIQUE NOT NULL,
  wellbeing_status VARCHAR(60) DEFAULT 'Thriving & Engaged'
);

CREATE TABLE timetable_slots (
  id SERIAL PRIMARY KEY,
  class_id VARCHAR(32) NOT NULL,
  day_of_week VARCHAR(16) NOT NULL,
  period_index INT CHECK (period_index BETWEEN 1 AND 9),
  subject VARCHAR(60) NOT NULL,
  teacher_id VARCHAR(32) REFERENCES teachers(id),
  original_teacher_id VARCHAR(32) REFERENCES teachers(id),
  room VARCHAR(32) NOT NULL,
  status VARCHAR(24) DEFAULT 'Regular',
  note TEXT
);

CREATE TABLE educational_materials (
  id VARCHAR(32) PRIMARY KEY,
  division VARCHAR(20) NOT NULL,
  class_id VARCHAR(32) NOT NULL,
  subject VARCHAR(60) NOT NULL,
  teacher_id VARCHAR(32) REFERENCES teachers(id),
  material_type VARCHAR(60) NOT NULL,
  title VARCHAR(200) NOT NULL,
  description TEXT,
  gems_reward INT DEFAULT 50,
  cognitive_pacing VARCHAR(60),
  completions_count INT DEFAULT 0,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
